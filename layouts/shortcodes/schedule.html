{{ .time }} {{ .break }}
{{- else }} {{- $headingText := printf "%s %s" .time .title }} {{- with .video }}{{ $headingText = printf "%s video" $headingText }}{{ end }}

"\n\n") }}{{ end }} {{ partial "shortcodes/expand.html" (dict "page" $page "title" $speakerTitle "content" $bioHTML) }} {{- $abstractHTML := " " }} {{- with .abstract }} {{- $abstractHTML = $page.RenderString (dict "display" "block") . }} {{- else }} {{- warnf "schedule %s: %q is missing an abstract (%s)" $year .title $page.File.Path }} {{- end }} {{ partial "shortcodes/expand.html" (dict "page" $page "title" "Abstract" "content" $abstractHTML) }} {{- end }} {{- end }}
